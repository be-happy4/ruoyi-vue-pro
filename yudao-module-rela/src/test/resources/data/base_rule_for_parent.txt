# 父母关系为基础的推导规则集

## 直系关系推导
(A.父亲=B OR A.母亲=B) AND A.sex="M" => B.儿子=A
(A.父亲=B OR A.母亲=B) AND A.sex="F" => B.女儿=A
A.父亲=C AND A.sex="M" AND B.父亲=C AND A.birthDay < B.birthDay => B.哥哥=A
A.父亲=C AND A.sex="F" AND B.父亲=C AND A.birthDay < B.birthDay => B.姐姐=A
A.父亲=C AND A.sex="M" AND B.父亲=C AND A.birthDay > B.birthDay => B.弟弟=A
A.父亲=C AND A.sex="F" AND B.父亲=C AND A.birthDay > B.birthDay => B.妹妹=A

A.父亲=B AND B.父亲=C => A.爷爷=C
A.父亲=B AND B.母亲=C => A.奶奶=C
A.母亲=B AND B.父亲=C => A.外公=C
A.母亲=B AND B.母亲=C => A.外婆=C

A.父亲=B AND A.sex="M" AND (B.父亲=C OR B.母亲=C) => C.孙子=A
A.父亲=B AND A.sex="F" AND (B.父亲=C OR B.母亲=C) => C.孙女=A
A.父亲=B AND B.父亲=C,A.sex="F" => A.爷爷=C,C.孙女=A
A.父亲=B AND B.母亲=C => A.奶奶=C
A.母亲=B AND B.父亲=C => A.外公=C
A.母亲=B AND B.母亲=C => A.外婆=C
